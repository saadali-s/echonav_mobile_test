  mapId: string;
}

export function FloorPlanCanvas({ mapId }: FloorPlanCanvasProps) {
  // 1. Zoom State (Controlled by the sleek buttons)
  const [zoomScale, setZoomScale] = useState(1);

  // 2. Pan Engine (Native physics for buttery smooth dragging)
  const pan = useRef(new Animated.ValueXY({ x: 0, y: 0 })).current;

  const panResponder = useRef(
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onPanResponderGrant: () => {
        // When dragging starts, extract the current offset so it doesn't snap back to 0
        pan.extractOffset();
      },
      onPanResponderMove: Animated.event(
        [null, { dx: pan.x, dy: pan.y }],
        { useNativeDriver: false } // layout animations don't support native driver on all RN versions
      ),
      onPanResponderRelease: () => {
        // Flatten the offset so the next drag starts from the correct position
        pan.flattenOffset();
      }
    })
  ).current;

  const handleZoomIn = () => setZoomScale(prev => Math.min(prev + 0.25, 3));
  const handleZoomOut = () => setZoomScale(prev => Math.max(prev - 0.25, 0.5));

  const gridLines = Array.from({ length: Math.floor(CANVAS_SIZE / GRID_STEP) + 1 });

  return (
    <View style={styles.wrapper}>
      
      {/* Top Left: Legend */}

interface RadarMapProps {
  isScanning: boolean;
  distance: number | null; // Fixed type to match useScanner
  heading: number;
}

export function RadarMap({ isScanning, distance, heading }: RadarMapProps) {
  const RADAR_SIZE = 260;
  const CENTER = RADAR_SIZE / 2;
  const [estimatedBearing, setEstimatedBearing] = useState<number | null>(null);
  const prevDistance = useRef<number | null>(null);

  useEffect(() => {
    if (!isScanning || distance === null) {
      setEstimatedBearing(null);

  }, [distance, heading, isScanning]);

  const radarRadius = (Math.max(0.1, Math.min(distance ?? 10, 10)) / 10) * (CENTER - 20);
  const angleRad = (estimatedBearing !== null ? (estimatedBearing - heading - 90) : 0) * (Math.PI / 180);
  
  const beaconX = CENTER + radarRadius * Math.cos(angleRad) - 10;
  const beaconY = CENTER + radarRadius * Math.sin(angleRad) - 10;

    <AppCard padded={false} style={styles.outerCard}>
      <View style={[styles.radarCircle, { width: RADAR_SIZE, height: RADAR_SIZE }]}>
