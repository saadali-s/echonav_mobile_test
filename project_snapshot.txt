import 'react-native-reanimated'; 
import '../global.css';
import { DarkTheme, DefaultTheme, ThemeProvider } from '@react-navigation/native';
import { Stack, useRouter, useSegments } from 'expo-router';
import { useColorScheme } from 'nativewind';
import React, { useEffect } from 'react';
import { Provider as PaperProvider } from 'react-native-paper';
import { AlertProvider } from '../core/context/AlertContext';
import { AuthProvider, useAuth } from '../core/context/AuthContext';

function RootLayoutNav() {
  const { isLoggedIn, role } = useAuth();
  const segments = useSegments();
  const router = useRouter();
  const { colorScheme } = useColorScheme();

  useEffect(() => {
    const inStaffGroup = segments[0] === 'staff';

    if (!isLoggedIn && inStaffGroup) {
      router.replace('/');
    } 
    else if (isLoggedIn && role !== 'NAV_USER' && (!segments[0] || (segments[0] as string) === '(user)')) {
      router.replace('/staff/(tabs)' as any);
    }
  }, [isLoggedIn, segments, role]);

  return (
    <ThemeProvider value={colorScheme === 'dark' ? DarkTheme : DefaultTheme}>
      <Stack screenOptions={{ headerShown: false }}>
        <Stack.Screen name="index" />
        <Stack.Screen name="staff" />
        <Stack.Screen name="navigate/index" />
      </Stack>
    </ThemeProvider>
  );
}

export default function RootLayout() {
  return (
    <AuthProvider>
      <AlertProvider>
        <PaperProvider>
          <RootLayoutNav />
        </PaperProvider>
      </AlertProvider>
    </AuthProvider>
  );
}

import { Feather } from '@expo/vector-icons';
import { useRouter } from 'expo-router';
import React, { useEffect, useRef, useState } from 'react';
import { KeyboardAvoidingView, Modal, Platform, ScrollView, TouchableOpacity, View } from 'react-native';
import { PairingModal } from '../components/auth/PairingModal';
import { RadarMap } from '../components/navigation/RadarMap';
import { AppButton } from '../components/ui/AppButton';
import { AppInput } from '../components/ui/AppInput';
import { AppScreen } from '../components/ui/AppScreen';
import { AppText } from '../components/ui/AppText';
import { BEACON_CONFIG } from '../core/constants/Config';
import { useAppAlert } from '../core/context/AlertContext';
import { useAuth } from '../core/context/AuthContext';
import { useScanner } from '../core/hooks/useScanner';
import { AuthAPI } from '../core/services/AuthAPI';
import { NavigationAPI } from '../core/services/NavigationAPI';

export default function RadarHomeScreen() {
  const router = useRouter();
  const { login } = useAuth();
  const { showAlert } = useAppAlert();
  
  const { startScanning, stopScanning, isScanning, distance, scannedDevicesList, bluetoothState } = useScanner();

  const [showAccess, setShowAccess] = useState(false);
  const [showStaffLogin, setShowStaffLogin] = useState(false);
  const [pendingRoute, setPendingRoute] = useState<{startNodeId: string, floorPlanId: string} | null>(null);
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  
  const isResolving = useRef(false);
  const scannedListRef = useRef<any[]>([]);

  useEffect(() => {
    scannedListRef.current = scannedDevicesList;
  }, [scannedDevicesList]);

  const handleActivateRadar = async () => {
    if (isResolving.current) return;
    isResolving.current = true;
    
    try {
      await startScanning();
    } catch (err: any) {
      showAlert('SCANNER OFFLINE', err.message, 'error');
      isResolving.current = false;
      return;
    }
    
    setTimeout(async () => {
      stopScanning();
      
      const topDevices = scannedListRef.current;
      
      if (topDevices.length > 0) {
        const closestBeacon = topDevices[0];

        try {
          const response = await NavigationAPI.resolvePosition(
            closestBeacon.uuid,
            BEACON_CONFIG.MAJOR,
            BEACON_CONFIG.MINOR
          );
          
          if (response.privacyLevel === 'PRIVATE') {
            setPendingRoute({ startNodeId: String(response.node.id), floorPlanId: String(response.floorPlanId) });
            setShowAccess(true);
          } else {
            router.push({ pathname: '/navigate', params: { startNodeId: String(response.node.id), floorPlanId: String(response.floorPlanId) } });
          }

        } catch (error: any) {
          showAlert("RESOLUTION FAILED", error.message || "Hardware signature not recognized by the server.", "error");
        }
      } else {
        showAlert("SCAN COMPLETE", "No spatial anchors found nearby.", "info");
      }
      
      isResolving.current = false;
    }, 2500);
  };

  const handleStaffLogin = async () => {
    if (!email || !password) return;
    try {
      setIsLoading(true);
      const response = await AuthAPI.loginStaff(email, password);
      
      if (!response || typeof response.token !== 'string') {
        throw new Error("Server returned an invalid authentication token.");
      }

      await login(response.token, response.userId);
      
      setShowStaffLogin(false);
      stopScanning();
      
      router.replace('/staff/(tabs)' as any);
    } catch (error: any) {
      showAlert("AUTHENTICATION FAILED", error.message || "Invalid credentials.", "error");
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <AppScreen padded={false} scrollable={false} centered>
      
      <View pointerEvents="box-none" className="absolute top-[60px] w-full px-6 flex-row justify-between items-center z-10">
        <TouchableOpacity 
          className="flex-row items-center bg-white dark:bg-zinc-900 px-4 py-2 rounded-full border-2 border-black dark:border-zinc-700 shadow-sm" 
          onPress={() => setShowAccess(true)}
        >
          <Feather name="key" size={16} className="text-emerald-600 dark:text-emerald-500 mr-2" />
          <AppText variant="label" color="primary">Bind</AppText>
        </TouchableOpacity>

        <TouchableOpacity 
          className="flex-row items-center bg-white dark:bg-zinc-900 px-4 py-2 rounded-full border-2 border-black dark:border-zinc-700 shadow-sm" 
          onPress={() => setShowStaffLogin(true)}
        >
          <AppText variant="label" color="primary" className="mr-2">Staff</AppText>
          <Feather name="shield" size={16} className="text-blue-600 dark:text-blue-500" />
        </TouchableOpacity>
      </View>

      <View className="w-full items-center mt-10">
        <AppText variant="displaySm" color="primary" center className="mb-4">
          EchoNav
        </AppText>
        
        <View className="min-h-[56px] w-[85%] mb-8 justify-center">
          {bluetoothState === 'PoweredOff' ? (
            <AppText variant="caption" color="danger" center>BLUETOOTH ANTENNA IS OFF</AppText>
          ) : isScanning ? (
            <AppText variant="caption" color="success" center>SEARCHING FOR SPATIAL ANCHORS...</AppText>
          ) : (
            <AppButton variant="primary" icon="radio" onPress={handleActivateRadar}>
              ACTIVATE SCANNER
            </AppButton>
          )}
        </View>

        <RadarMap isScanning={isScanning} distance={distance} heading={0} />
      </View>

      <PairingModal 
        visible={showAccess}
        onClose={() => { setShowAccess(false); isResolving.current = false; setPendingRoute(null); }} 
        onAuthenticate={async (accEmail, pin) => {
          try {
            const response = await AuthAPI.pairDevice(accEmail, pin);
            await login(response.token, response.userId);
            setShowAccess(false);
            
            if (pendingRoute) {
               router.replace({ pathname: '/navigate', params: pendingRoute } as any);
            } else {
               showAlert("SUCCESS", "Device paired, but no anchor was scanned to route to.", "success");
            }
          } catch (error: any) {
            showAlert("PAIRING FAILED", error.message || "Invalid PIN or Email.", "error");
          }
        }}
      />

      <Modal visible={showStaffLogin} transparent animationType="fade">
        <View className="flex-1 bg-black/70">
          {/* THE FIX: behavior="padding" universally + isolated outer View */}
          <KeyboardAvoidingView 
            behavior="padding" 
            className="flex-1"
          >
            <ScrollView 
              contentContainerStyle={{ flexGrow: 1, justifyContent: 'center', padding: 20 }}
              keyboardShouldPersistTaps="handled"
              showsVerticalScrollIndicator={false}
            >
              <View className="bg-white dark:bg-zinc-950 p-6 rounded-2xl border-2 border-black dark:border-zinc-800">
                <View className="flex-row justify-between items-center mb-5">
                  <AppText variant="titleMd">Staff Access</AppText>
                  <TouchableOpacity onPress={() => setShowStaffLogin(false)}>
                    <Feather name="x" size={24} className="text-black dark:text-white" />
                  </TouchableOpacity>
                </View>

                <AppInput label="Email" value={email} onChangeText={setEmail} autoCapitalize="none" keyboardType="email-address" />
                <AppInput label="Password" value={password} onChangeText={setPassword} secureTextEntry />
                <AppButton onPress={handleStaffLogin} loading={isLoading} className="mt-3">
                  AUTHORIZE
                </AppButton>
              </View>
            </ScrollView>
          </KeyboardAvoidingView>
        </View>
      </Modal>

    </AppScreen>
  );
}

import { AppScreen } from '@/components/ui/AppScreen';
import { useLocalSearchParams, useRouter } from 'expo-router';
import React from 'react';
import { ActivityIndicator } from 'react-native';

import { DirectoryView } from '@/components/navigation/DirectoryView';
import { NavigatingView } from '@/components/navigation/NavigatingView';

import { useNavigationController } from '@/core/hooks/useNavigationController';

export default function NavigateScreen() {
  const router = useRouter();
  const { startNodeId, floorPlanId } = useLocalSearchParams();

  const {
    navState, isLoading, allNodes, currentNodeInfo, activeRoute, currentTargetIndex,
    isScanning, distance, isSpinning, triggerAudio, handleSelectDestination,
    handleCancelRoute, directoryAudioMap
  } = useNavigationController(startNodeId as string, floorPlanId as string);

  if (isLoading) {
    return (
      <AppScreen centered padded={false}>
        <ActivityIndicator size="large" color="#10b981" />
      </AppScreen>
    );
  }

  if (navState === 'NAVIGATING' && activeRoute) {
    const targetNode = activeRoute.path[currentTargetIndex];
    const pathLabels = activeRoute.path.map((n: any) => n.label || `Node ${n.id}`);

    return (
      <NavigatingView 
        targetNodeLabel={targetNode?.label || "DESTINATION"}
        isScanning={isScanning}
        distance={distance}
        isSpinning={isSpinning}
        totalDistanceMeters={activeRoute.totalDistanceMeters}
        estimatedTimeSeconds={activeRoute.estimatedTimeSeconds}
        pathLabels={pathLabels}
        onTriggerAudio={triggerAudio}
        onCancel={handleCancelRoute}
      />
    );
  }

  return (
    <DirectoryView 
      currentNodeLabel={currentNodeInfo?.label || `Node ${startNodeId}`}
      startNodeId={startNodeId as string}
      allNodes={allNodes}
      directoryAudioMap={directoryAudioMap}
      onClose={() => router.replace('/')}
      onSelectDestination={handleSelectDestination}
    />
  );
}

import { useRouter } from 'expo-router';

import React, { useEffect, useState } from 'react';

import { CompactTaskCard } from '../../../components/fleet/CompactTaskCard';
import { CompletedTaskTag } from '../../../components/fleet/CompletedTaskTag';
import { FloorPlanCard } from '../../../components/maps/FloorPlanCard';
import { StaffHeader } from '../../../components/staff/StaffHeader';
import { AppScreen } from '../../../components/ui/AppScreen';
import { AppText } from '../../../components/ui/AppText';
import { useAppAlert } from '../../../core/context/AlertContext';
import { useAuth } from '../../../core/context/AuthContext';
import { DashboardAPI } from '../../../core/services/DashboardAPI';
import { FleetAPI } from '../../../core/services/FleetAPI';

export default function StaffHome() {
  const router = useRouter();
  const { colorScheme } = useColorScheme();
  const { orgId, userId, logout } = useAuth();
  const { showAlert } = useAppAlert();

  const [isLoading, setIsLoading] = useState(true);
  const [isRefreshing, setIsRefreshing] = useState(false);
  const [dashboardData, setDashboardData] = useState<any>(null);
  const [tasks, setTasks] = useState<any[]>([]);

  const fetchDashboardData = async () => {
    if (!orgId || !userId) return;
    try {
      const [dashRes, tasksRes] = await Promise.all([
        DashboardAPI.getDashboardData(orgId),
        FleetAPI.getStaffTasks(userId)
      ]);
      setDashboardData(dashRes);
      setTasks(tasksRes);
    } catch (error: any) {
      showAlert("SYNC ERROR", error.message || "Failed to load dashboard data.", "error");
    }
  };

  useEffect(() => {
    const initLoad = async () => {
      setIsLoading(true);
      await fetchDashboardData();
      setIsLoading(false);
    };
    initLoad();
  }, [orgId, userId]);

  const handleRefresh = async () => {
    setIsRefreshing(true);
    await fetchDashboardData();
    setIsRefreshing(false);
  };

  const handleLogout = async () => {
    await logout();
    router.replace('/');
  };

  if (isLoading || !dashboardData) {
    return (
      <AppScreen centered padded={false}>
        <ActivityIndicator size="large" color={colorScheme === 'dark' ? '#10b981' : '#000'} />
      </AppScreen>
    );
  }

  const completedTasks = tasks.filter(t => t.status === 'VERIFIED' || t.status === 'COMPLETED');
  const pendingTasks = tasks.filter(t => t.status === 'TO_DO' || t.status === 'PENDING' || t.status === 'IN_PROGRESS').slice(0, 4);

  return (
    <AppScreen padded={false} scrollable={false}>
      <StaffHeader name={dashboardData.userName || "Operator"} onLogout={handleLogout} />
      
      <ScrollView 
        showsVerticalScrollIndicator={false} 
        contentContainerStyle={{ paddingBottom: 100 }}
        refreshControl={<RefreshControl refreshing={isRefreshing} onRefresh={handleRefresh} tintColor="#10b981" colors={['#10b981']} />}
      >
        
        {completedTasks.length > 0 && (
          <View className="mt-5 mb-2">
            <View className="px-5 mb-2">
              <AppText variant="label" color="muted">Recently Completed</AppText>
            </View>
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ paddingHorizontal: 20, gap: 8 }}>
              {completedTasks.map(t => (
                <CompletedTaskTag key={t.id} title={`Node #${t.nodeLabel}`} />
              ))}
            </ScrollView>
          </View>
        )}

        <View className="px-5 mt-4">
          <View className="flex-row justify-between items-baseline mb-3">

import { useRouter } from 'expo-router';
import { useColorScheme } from 'nativewind';
import React, { useEffect, useState } from 'react';
import { ActivityIndicator, FlatList, RefreshControl, View } from 'react-native';
import { FloorPlanCard } from '../../../components/maps/FloorPlanCard';
import { AppScreen } from '../../../components/ui/AppScreen';
import { AppText } from '../../../components/ui/AppText';
import { SearchInput } from '../../../components/ui/SearchInput';
import { useAppAlert } from '../../../core/context/AlertContext';
import { useAuth } from '../../../core/context/AuthContext';
import { MapStudioAPI } from '../../../core/services/MapStudioAPI';

export default function MapsScreen() {
  const router = useRouter();
  const { orgId } = useAuth();
  const { showAlert } = useAppAlert();
  const { colorScheme } = useColorScheme();
  
  const [search, setSearch] = useState('');
  const [maps, setMaps] = useState<any[]>([]);
  
  const [isLoading, setIsLoading] = useState(true);
  const [isRefreshing, setIsRefreshing] = useState(false);

  const fetchAllMaps = async () => {
    if (!orgId) return;
    try {
      const buildings = await MapStudioAPI.getBuildings(orgId);
      
      const floorPlanPromises = buildings.map(async (b: any) => {
        const plans = await MapStudioAPI.getFloorPlans(b.id);
        return plans.map((p: any) => ({
          id: String(p.id),
          title: p.name || 'Unnamed Floor',
          building: b.name || 'Unknown Building'
        }));
      });

      const results = await Promise.all(floorPlanPromises);
      setMaps(results.flat());
    } catch (error: any) {
      showAlert("NETWORK ERROR", error.message || "Failed to load map directory.", "error");
    }
  };

  useEffect(() => {
    const initLoad = async () => {
      setIsLoading(true);
      await fetchAllMaps();
      setIsLoading(false);
    };
    initLoad();
  }, [orgId]);

  const handleRefresh = async () => {
    setIsRefreshing(true);
    await fetchAllMaps();
    setIsRefreshing(false);
  };

  const filteredMaps = maps.filter(m => 
    m.title.toLowerCase().includes(search.toLowerCase()) || 
    m.building.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <AppScreen padded={false} scrollable={false}>
      <View className="p-5 bg-white dark:bg-zinc-950 border-b-[3px] border-black dark:border-zinc-800">
        <AppText variant="displaySm" className="mb-3">Site Explorer</AppText>
        <SearchInput 
          value={search} 
          onChangeText={setSearch} 
          placeholder="Filter by building or level..." 
        />
      </View>

      {isLoading ? (
        <View className="flex-1 justify-center items-center">
          <ActivityIndicator size="large" color={colorScheme === 'dark' ? '#10b981' : '#000'} />
          <AppText variant="caption" color="muted" className="mt-4">DOWNLOADING VECTORS...</AppText>
        </View>
      ) : (
        <FlatList
          data={filteredMaps}
          numColumns={2}
          keyExtractor={(item) => item.id}
          columnWrapperStyle={{ justifyContent: 'space-between', paddingHorizontal: 20 }}
          contentContainerStyle={{ paddingVertical: 20, paddingBottom: 100 }}
          refreshControl={<RefreshControl refreshing={isRefreshing} onRefresh={handleRefresh} tintColor="#10b981" colors={['#10b981']} />}

import { AppButton } from '@/components/ui/AppButton';
import { AppCard } from '@/components/ui/AppCard';

import { Stack, useLocalSearchParams, useRouter } from 'expo-router';
import React, { useEffect, useRef, useState } from 'react';
import { Vibration, View } from 'react-native'; 

export default function CalibrationScreen() {
  const { id } = useLocalSearchParams();
  const router = useRouter();
  const { showAlert } = useAppAlert();
  
  const { rssi: rawRssi, startScanning, stopScanning, bluetoothState } = useScanner();

  const [isCalibrating, setIsCalibrating] = useState(false);
  const [progress, setProgress] = useState(0);
  const [baselineRssi, setBaselineRssi] = useState<number | null>(null);
  const [isSaving, setIsSaving] = useState(false); 
  
  const samplesRef = useRef<number[]>([]);

  useEffect(() => {
    if (isCalibrating && rawRssi !== null) {
      samplesRef.current.push(rawRssi);
    }
  }, [rawRssi, isCalibrating]);

  const initiateCalibrationSequence = async () => {
    try {
      await startScanning();
    } catch (err: any) {
      showAlert('HARDWARE ERROR', err.message, 'error');
      return;
    }

    Vibration.vibrate(100);
    setIsCalibrating(true);
    setProgress(0);
    setBaselineRssi(null);
    samplesRef.current = [];

    let currentProgress = 0;
    const interval = setInterval(() => {
      currentProgress += 2; 
      setProgress(currentProgress);

      if (currentProgress >= 100) {
        clearInterval(interval);
        stopScanning();
        setIsCalibrating(false);
        Vibration.vibrate([0, 200, 100, 200]); 
        
        const samples = samplesRef.current;
        if (samples.length > 0) {
          const sum = samples.reduce((a, b) => a + b, 0);
          setBaselineRssi(Math.round(sum / samples.length));
        } else {
          setBaselineRssi(-65);
        }
      }
    }, 100);
  };

  const handleComplete = async () => {
    if (!baselineRssi) return;

    try {
      setIsSaving(true);
      await FleetAPI.updateTaskStatus(id as string, 'VERIFIED', baselineRssi);
      
      showAlert("SUCCESS", "Beacon calibrated and deployed successfully.", "success", "FINISH", () => {
         router.replace('/staff/(tabs)' as any);
      });
    } catch (error: any) {
      showAlert("COMMIT ERROR", error.message || "Failed to save the baseline to the database.", "error");
    } finally {
      setIsSaving(false);
    }
  };

  if (bluetoothState === 'PoweredOff') {
    return (
      <AppScreen centered>
         <AppText variant="displaySm" color="danger">BLUETOOTH OFF</AppText>
         <AppText variant="body" color="muted" center className="mt-3">
           Please enable Bluetooth to continue calibration.
         </AppText>
      </AppScreen>
    );
  }

  return (
    <AppScreen>
      <Stack.Screen options={{ title: 'SIGNAL CALIBRATION', headerShown: true }} />

      <View className="mb-8 items-center">
        <View className="w-20 h-20 rounded-full bg-zinc-100 dark:bg-zinc-900 justify-center items-center border-2 border-black dark:border-zinc-700 mb-4">
          <Feather name="crosshair" size={32} className="text-black dark:text-white" />
        </View>
        <AppText variant="displayMd" center>Baseline Setup</AppText>
        <AppText variant="bodySm" color="secondary" center className="mt-1">
          Stand exactly 1 meter away from the beacon at chest height.
        </AppText>
      </View>

      <AppCard variant={baselineRssi !== null ? 'success' : 'default'}>
        {baselineRssi === null ? (
          <View className="gap-5">
            <View>
              <AppText variant="titleSm" center>Calibration Status</AppText>
              <View className="h-3 bg-zinc-200 dark:bg-zinc-800 rounded-full mt-3 overflow-hidden border border-zinc-300 dark:border-zinc-700">
                <View style={{ height: '100%', width: `${progress}%` }} className="bg-emerald-500 dark:bg-emerald-400" />
              </View>
              <AppText variant="caption" color="secondary" center className="mt-2">
                {isCalibrating ? `SAMPLING DATA... ${progress}%` : 'READY TO START'}
              </AppText>
            </View>

            <AppButton 
              onPress={initiateCalibrationSequence} 
              disabled={isCalibrating}
              loading={isCalibrating}
            >
              START 5-SECOND SAMPLE
            </AppButton>
          </View>
        ) : (
          <View className="gap-4 items-center">
            <Feather name="check-circle" size={48} className="text-emerald-700 dark:text-emerald-500" />
            <View className="items-center">
              <AppText variant="titleMd" color="success">CALIBRATION SUCCESSFUL</AppText>
              <AppText variant="displayLg" className="mt-2">
                {baselineRssi} <AppText variant="titleSm">dBm</AppText>
              </AppText>
              <AppText variant="caption" color="secondary">
                Calculated from {samplesRef.current.length} samples
              </AppText>
            </View>
            
            <AppButton 
              variant="primary" 
              onPress={handleComplete}
              loading={isSaving}
              disabled={isSaving}
            >
              SAVE & COMPLETE TASK
            </AppButton>
          </View>
        )}
      </AppCard>

      <View className="mt-5 px-3">
        <AppText variant="caption" color="muted" center>
          Ensure no physical obstructions are between the phone and the beacon during sampling.
        </AppText>
      </View>
    </AppScreen>
  );

import { Feather } from '@expo/vector-icons';
import { Audio } from 'expo-av';
import React, { useEffect, useRef, useState } from 'react';
import { PanResponder, TouchableOpacity, Vibration, View } from 'react-native';
import { AppScreen } from '../ui/AppScreen';
import { AppText } from '../ui/AppText';

export function DirectoryView({ 
  currentNodeLabel, startNodeId, allNodes, directoryAudioMap, onClose, onSelectDestination 
}: any) {
  const [selectedIndex, setSelectedIndex] = useState(0);

  const soundRef = useRef<Audio.Sound | null>(null);
  const lastDy = useRef(0);
  const accumulatedDy = useRef(0);
  const lastTapTime = useRef(0);

  useEffect(() => {
    if (allNodes.length === 0) return;
    const timer = setTimeout(async () => {
      if (soundRef.current) {
        await soundRef.current.unloadAsync();
        soundRef.current = null;
      }
      
      const selectedNode = allNodes[selectedIndex];
      const audioUri = directoryAudioMap[String(selectedNode.id)];
      
      if (audioUri) {
        try {
          const { sound } = await Audio.Sound.createAsync({ uri: audioUri });
          soundRef.current = sound;
          await sound.playAsync();
        } catch (e) {
          console.error("Dial TTS Error", e);
        }
      }
    }, 200); 
    return () => clearTimeout(timer);
  }, [selectedIndex, allNodes, directoryAudioMap]);

  useEffect(() => {
    return () => {
      if (soundRef.current) soundRef.current.unloadAsync();
    };
  }, []);

  const panResponder = useRef(
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onMoveShouldSetPanResponder: () => true,
      onPanResponderGrant: () => { lastDy.current = 0; accumulatedDy.current = 0; },
      onPanResponderMove: (_, gestureState) => {
        const delta = gestureState.dy - lastDy.current;
        lastDy.current = gestureState.dy;
        accumulatedDy.current += delta;
        
        if (accumulatedDy.current < -35) {
          setSelectedIndex(prev => {
            const next = Math.min(prev + 1, allNodes.length - 1);
            if (next !== prev) Vibration.vibrate(10);
            return next;
          });
          accumulatedDy.current = 0;
        } else if (accumulatedDy.current > 35) {
          setSelectedIndex(prev => {
            const prevIdx = Math.max(prev - 1, 0);
            if (prevIdx !== prev) Vibration.vibrate(10);
            return prevIdx;
          });
          accumulatedDy.current = 0;
        }
      },
      onPanResponderRelease: (_, gestureState) => {
        if (Math.abs(gestureState.dx) < 10 && Math.abs(gestureState.dy) < 10) {
          const now = Date.now();
          if (now - lastTapTime.current < 400) {
            handleDoubleTap();
            lastTapTime.current = 0; 
          } else {
            lastTapTime.current = now;
          }
        }
      }
    })
  ).current;

  const handleDoubleTap = () => {
    if (allNodes.length === 0) return;
    const selectedNode = allNodes[selectedIndex];
    
    if (String(selectedNode.id) === startNodeId) {
      Vibration.vibrate([0, 100, 50, 100]);
    } else {
      Vibration.vibrate(100);
      onSelectDestination(String(selectedNode.id));
    }
  };

  const renderRotaryItems = () => {
    return allNodes.map((node: any, i: number) => {
      const offset = i - selectedIndex;
      if (Math.abs(offset) > 3) return null;

      const isCenter = offset === 0;
      const isCurrentLocation = String(node.id) === startNodeId;

      return (
        <View 
            key={node.id} 
            className="absolute w-[85%] p-5 rounded-2xl items-center justify-center"
            style={{ 
                transform: [
                    { translateY: offset * 80 }, 
                    { scale: Math.max(0.6, 1 - Math.abs(offset) * 0.2) }
                ],
                opacity: Math.max(0.1, 1 - Math.abs(offset) * 0.4),
                borderColor: isCenter ? '#10b981' : 'transparent',
                borderWidth: isCenter ? 2 : 0,
                // THE FIX: Dial background updates properly in dark mode
                backgroundColor: isCenter ? 'rgba(128,128,128,0.1)' : 'transparent'
            }}
        >
            <View className="items-center">
                <AppText variant={isCenter ? "displaySm" : "titleMd"} color={isCenter ? "success" : "primary"} center>
                    {node.label || `Node ${node.id}`}
                </AppText>
                
                {isCurrentLocation && isCenter && (
                    <AppText variant="caption" color="muted" className="mt-2">YOU ARE HERE</AppText>
                )}
            </View>
        </View>
      );
    });
  };

  return (
    <AppScreen padded={false} scrollable={false}>
      
      <View className="p-5 pt-[60px] bg-white dark:bg-[#111] border-b border-zinc-200 dark:border-[#333] z-10">
        <View className="flex-row justify-between items-center mb-3">
            <AppText variant="titleMd" color="primary">Directory Dial</AppText>
            <TouchableOpacity onPress={onClose}>
                <Feather name="x" size={24} className="text-zinc-400" />
            </TouchableOpacity>
        </View>
        <View className="flex-row items-center gap-2">
            <Feather name="map-pin" size={16} className="text-emerald-500" />
            <AppText variant="bodySm" color="muted">
                Current Location: <AppText color="primary">{currentNodeLabel}</AppText>
            </AppText>
        </View>
      </View>

      <View className="flex-1 bg-zinc-50 dark:bg-black justify-center overflow-hidden" {...panResponder.panHandlers}>
        <View className="absolute top-8 w-full items-center z-10">
            <AppText variant="label" color="muted" center>SWIPE ANYWHERE TO SPIN DIAL</AppText>
            <AppText variant="caption" color="muted" center>DOUBLE-TAP TO CONFIRM</AppText>
        </View>
        
        <View className="h-[100px] justify-center items-center">
            {renderRotaryItems()}
        </View>
      </View>

    </AppScreen>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { View } from 'react-native';
import { AppCard } from '../ui/AppCard';
import { AppText } from '../ui/AppText';

interface RadarMapProps {
  isScanning: boolean;
  distance: number | null; 
  heading: number;
}

export function RadarMap({ isScanning, distance, heading }: RadarMapProps) {
  const RADAR_SIZE = 260;
  const CENTER = RADAR_SIZE / 2;
  const [estimatedBearing, setEstimatedBearing] = useState<number | null>(null);
  const prevDistance = useRef<number | null>(null);

  useEffect(() => {
    if (!isScanning || distance === null) {
      setEstimatedBearing(null);
      prevDistance.current = null;
      return;
    }

    if (estimatedBearing === null) {
      setEstimatedBearing(heading);
    } else if (prevDistance.current !== null) {
      const delta = distance - prevDistance.current;
      if (delta < -0.2) setEstimatedBearing(heading);
      else if (delta > 0.2) setEstimatedBearing((heading + 180) % 360);
    }
    prevDistance.current = distance;
  }, [distance, heading, isScanning]);

  const radarRadius = (Math.max(0.1, Math.min(distance ?? 10, 10)) / 10) * (CENTER - 20);
  const angleRad = (estimatedBearing !== null ? (estimatedBearing - heading - 90) : 0) * (Math.PI / 180);
  
  const beaconX = CENTER + radarRadius * Math.cos(angleRad) - 10;
  const beaconY = CENTER + radarRadius * Math.sin(angleRad) - 10;

  return (
    <AppCard padded={false} className="bg-zinc-50 dark:bg-black items-center overflow-hidden border-zinc-200 dark:border-zinc-800">
      <View className="relative items-center justify-center rounded-full bg-white dark:bg-[#0a0a0a] my-5 shadow-sm dark:shadow-none" style={{ width: RADAR_SIZE, height: RADAR_SIZE }}>

import React, { createContext, useContext, useEffect, useState } from 'react';

const decodeJWT = (token: string) => {
  try {
    const base64Url = token.split('.')[1];
    const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=';
    let str = base64.replace(/=+$/, '');
    let output = '';
    
    for (let bc = 0, bs = 0, buffer, idx = 0; (buffer = str.charAt(idx++)); ~buffer && (bs = bc % 4 ? bs * 64 + buffer : buffer, bc++ % 4) ? output += String.fromCharCode(255 & bs >> (-2 * bc & 6)) : 0) {
      buffer = chars.indexOf(buffer);
    }
    return JSON.parse(output);
  } catch (e) {
    console.error("JWT Decode Error", e);
    return null;
  }
};

interface AuthState {
  isLoggedIn: boolean;
  token: string | null;
  role: string | null;
  orgId: string | null;
  userId: string | null;
}

interface AuthContextType extends AuthState {
  login: (token: string, explicitUserId?: string | number) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType | null>(null);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [authState, setAuthState] = useState<AuthState>({
    isLoggedIn: false,
    token: null,
    role: null,
    orgId: null,
    userId: null,
  });

  useEffect(() => {
    const loadToken = async () => {
      const storedToken = await ApiClient.getToken();
      if (storedToken) {
        await login(storedToken);
      }
    };
    loadToken();
  }, []);

  const login = async (token: string, explicitUserId?: string | number) => {
    await ApiClient.setToken(token);
    const decoded = decodeJWT(token);
    
    // THE FIX: Pulls the user_id from the new JWT claim if an explicit one isn't passed
    const finalUserId = explicitUserId ? String(explicitUserId) : (decoded?.user_id ? String(decoded.user_id) : null);
    
    setAuthState({
      isLoggedIn: true,
      token,
      role: decoded?.role || null,
      orgId: decoded?.organization_id ? String(decoded.organization_id) : null,
      userId: finalUserId, 
    });
  };

  const logout = async () => {
    await ApiClient.clearToken();
    setAuthState({ isLoggedIn: false, token: null, role: null, orgId: null, userId: null });
  };

  return <AuthContext.Provider value={{ ...authState, login, logout }}>{children}</AuthContext.Provider>;
}

import { Gyroscope } from 'expo-sensors';
import { useEffect, useRef, useState } from 'react';

export const useKinematics = (currentSteps: number) => {
  const [isSpinning, setIsSpinning] = useState(false);
  const [hasTurnedAround, setHasTurnedAround] = useState(false);

  const stepsRef = useRef(currentSteps);
  const historyRef = useRef<{ timestamp: number; zRate: number; steps: number }[]>([]);

  useEffect(() => {
    stepsRef.current = currentSteps;
  }, [currentSteps]);

  useEffect(() => {
    let subscription: any;

    const startKinematics = async () => {
      const isAvailable = await Gyroscope.isAvailableAsync();
      if (!isAvailable) return;

      // 10 Hz sampling rate
      Gyroscope.setUpdateInterval(100); 

      subscription = Gyroscope.addListener((data) => {
        const now = Date.now();
        const currentStepsVal = stepsRef.current;

        // Focus purely on Z-axis (Yaw) angular velocity for horizontal turning
        const zRate = data.z;

        historyRef.current.push({ timestamp: now, zRate, steps: currentStepsVal });

        // Keep only the last 1.5 seconds of telemetry
        historyRef.current = historyRef.current.filter((entry) => now - entry.timestamp <= 1500);

        if (historyRef.current.length > 0) {
          // Integrate Z-axis rate over time (100ms = 0.1s) to compute turn degrees
          const totalAngleRads = historyRef.current.reduce((sum, entry) => sum + entry.zRate * 0.1, 0);
          const totalAngleDeg = Math.abs(totalAngleRads * (180 / Math.PI));

          const oldestSteps = historyRef.current[0].steps;
          const stepsTakenInWindow = currentStepsVal - oldestSteps;

          // 1. The Spin-In-Place Lock: Rotated > 45 degrees, but took 0 steps
          if (totalAngleDeg > 45 && stepsTakenInWindow === 0) {
            setIsSpinning(true);
          } else {
            setIsSpinning(false);
          }

          // 2. The 180-Degree About-Face: Rotated > 140 degrees WHILE walking
          if (totalAngleDeg > 140 && stepsTakenInWindow > 0) {
            setHasTurnedAround(true);
            historyRef.current = [];
          }
        }
      });
    };

    startKinematics();

    return () => {
      if (subscription) subscription.remove();
    };
  }, []);

  const clearTurnWarning = () => setHasTurnedAround(false);

  return { isSpinning, hasTurnedAround, clearTurnWarning };
};

    }
  };

  const handleSelectDestination = async (destinationNodeId: string) => {
    if (String(destinationNodeId) === String(startNodeId)) {
        Vibration.vibrate([0, 100, 50, 100]); 
        return; 
    }

    Vibration.vibrate(100);
    setIsLoading(true);
    try {
      const routeResponse = await NavigationAPI.getRoute(floorPlanId, startNodeId, destinationNodeId);
      
      if (routeResponse.status === "SUCCESS" && routeResponse.path && routeResponse.path.length > 0) {
         setActiveRoute(routeResponse);
         setCurrentTargetIndex(1); 
         
         const updatedAudioMap = await prefetchRouteAudio(routeResponse.path);
         
         setNavState('NAVIGATING');
         
         try {
            await startScanning();
         } catch (err: any) {
            showAlert('HARDWARE ERROR', err.message, 'error');
         }

      Vibration.vibrate([0, 200, 100, 200]);
      showAlert("Network Error", e.message || "Failed to calculate the route. Please try again.", "error");
    } finally {
      setIsLoading(false);
    }
  };

    isLoading,
    allNodes,
    currentNodeInfo,
    activeRoute,

import { useCallback, useRef } from 'react';

// The "Speed" of the filter.
const PROCESS_NOISE = 0.8; 

// How much we trust a signal that is GETTING STRONGER (Approaching). 
const MEASUREMENT_NOISE_APPROACH = 0.01; 

// How much we trust a signal that is GETTING WEAKER (Departing).
const MEASUREMENT_NOISE_DEPART = 5.0;    

export const useRSSIKalman = () => {
  const x = useRef<number>(0); 
  const p = useRef<number>(0); 
  const initialized = useRef<boolean>(false);

  const filter = useCallback((rssi: number): number => {
    if (!initialized.current) {
      x.current = rssi;
      p.current = 1.0;
      initialized.current = true;
      return rssi;
    }

    // THE FIX: Removed the 12dBm snapping logic. 
    // Human bodies block 2.4GHz signals instantly. The snap caused erratic map teleportation.
    // We now let the Kalman filter smooth out the body-blocks naturally.

    const R = rssi > x.current ? MEASUREMENT_NOISE_APPROACH : MEASUREMENT_NOISE_DEPART;

    const p_pred = p.current + PROCESS_NOISE;
    const K = p_pred / (p_pred + R);
    x.current = x.current + K * (rssi - x.current);
    p.current = (1 - K) * p_pred;

    return x.current;
  }, []);

  const reset = useCallback(() => {
    x.current = 0;
    p.current = 0;
    initialized.current = false;
  }, []);

  return { filter, reset };
};

import * as SecureStore from 'expo-secure-store';

const BASE_URL = process.env.EXPO_PUBLIC_API_URL || "http://192.168.3.112:8080/api"; 
const TOKEN_KEY = 'echonav_mobile_jwt';

export class ApiClient {
  
  static async setToken(jwt: string) {
    await SecureStore.setItemAsync(TOKEN_KEY, jwt);
  }

  static async clearToken() {
    await SecureStore.deleteItemAsync(TOKEN_KEY);
  }

  static async getToken(): Promise<string | null> {
    return await SecureStore.getItemAsync(TOKEN_KEY);
  }

  private static async request<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
    const url = `${BASE_URL}${endpoint}`;
    
    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      'X-EchoNav-App-Key': process.env.EXPO_PUBLIC_APP_KEY || 'default_dev_app_key_12345',
      ...(options.headers as Record<string, string> || {}),
    };

    const token = await this.getToken();
    if (token) {
      headers['Authorization'] = `Bearer ${token}`;
    }

    // THE FIX: 8-Second Execution Timeout
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 8000);

    try {
      const response = await fetch(url, { ...options, headers, signal: controller.signal as any });
      clearTimeout(timeoutId);
      
      const isJson = response.headers.get('content-type')?.includes('application/json');
      const data = isJson ? await response.json() : await response.text();

      if (!response.ok) {
        throw new Error(data.message || data.error || `HTTP Error ${response.status}`);
      }

      return data as T;
    } catch (error: any) {
      clearTimeout(timeoutId);
      if (error.name === 'AbortError') {
        throw new Error("Connection timed out. The server is unreachable.");
      }
      console.error(`[API ERROR] ${options.method || 'GET'} ${endpoint}:`, error.message);
      throw error;
    }
  }

  static async get<T>(endpoint: string) {
    return this.request<T>(endpoint, { method: 'GET' });
  }

  static async post<T>(endpoint: string, body: any) {
    return this.request<T>(endpoint, { method: 'POST', body: JSON.stringify(body) });
  }

  static async put<T>(endpoint: string, body: any) {
    return this.request<T>(endpoint, { method: 'PUT', body: JSON.stringify(body) });
  }
}
